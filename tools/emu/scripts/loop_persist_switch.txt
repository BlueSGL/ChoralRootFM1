# test_persist.sh, on the same flash (slot 1: G, 1 event): recorded over with three chords (D, Em, A: 3 events),
# stopped (dump 1: not saved); the PLAY menu's PRESETS +1: slot 2: the loop left is written to its slot (the one write
# that needs no SAVE held: "loop 2 . saved 1"); PRESETS -1: back, 3 events (dump 2). Quit without a SAVE held.
# loop_persist_back.txt: slot 1 has the 3 events.
wait 300
btn REC
wait 300
btn A#3 down
btn D4 down
wait 400
btn D4 up
btn A#3 up
wait 200
btn G#3 down
btn E4 down
wait 400
btn E4 up
btn G#3 up
wait 200
btn A#3 down
btn A4 down
wait 400
btn A4 up
btn A#3 up
wait 300
btn REC
wait 400
btn PLAY
wait 500
dump                          # 1: 3 events, not saved
btn PLAY down
wait 500
btn PLAY up
wait 300
shot build/emu/test/loop_persist_dirty.ppm    # the PLAY menu: the mark on slot 1 (not saved)
knob PRESETS +1
wait 300
shot build/emu/test/loop_persist_switch.ppm
knob PRESETS -1
wait 1500
shot build/emu/test/loop_persist_clean.ppm    # back on slot 1: saved, no mark
btn OCT-
wait 300
dump                          # 2: slot 1 again, 3 events, saved
quit
